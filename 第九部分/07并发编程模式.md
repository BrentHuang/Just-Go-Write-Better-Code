## 协程并发模式

### 生产者 - 消费者模式

生产者和消费者之间通过通道进行解耦，解决生产者和消费者速度不匹配的问题。

单个生产者和单个消费者，生产者异步生产完后关闭任务队列、退出，消费者异步处理完所有任务后，发送一个完成信号给主 Goroutine，然后退出。主 Goroutine 等待接收消费者的完成信号，再退出。

```go
type Task struct {
 ID      int
 Content string
 Handler func(t *Task) error
}

func (t Task) String() string {
 return fmt.Sprintf("id:%d,content:%s", t.ID, t.Content)
}

func NewTask(id int) Task {
 return Task{
  ID:      id,
  Content: strconv.Itoa(id),
  Handler: func(t *Task) error {
   // time.Sleep(time.Duration(rand.IntN(1000)) * time.Millisecond) // 模拟业务逻辑

   if rand.IntN(2) == 0 {
    return fmt.Errorf("模拟失败")
   }

   if rand.IntN(2) == 0 {
    panic("模拟 panic")
   }

   return nil
  },
 }
}

type ProducerConsumer struct {
 tasks         chan Task     // 生产者和消费者共享一个通道，不需要额外加锁。Task 结构很小，没有必要用指针类型
 consumerDone  chan struct{} // 消费者退出时发送 done 信号
 consumedCount atomic.Int64  // 消费者消费的任务总数，简单对一下帐
}

func NewProducerConsumer(bufSize int) *ProducerConsumer {
 return &ProducerConsumer{
  tasks:        make(chan Task, bufSize),
  consumerDone: make(chan struct{}),
 }
}

// 启动一个异步消费者
func (pc *ProducerConsumer) StartConsumer() {
 go func() {
  for task := range pc.tasks {
   pc.handleTask(&task)
   pc.consumedCount.Add(1)
  }

  fmt.Printf("消费者退出了，消费总数：%d\n", pc.consumedCount.Load())
  close(pc.consumerDone)
 }()
}

func (pc *ProducerConsumer) handleTask(task *Task) {
 defer func() {
  // 当 task 的 handler 中发生 panic 时，不至于让消费者协程退出
  if r := recover(); r != nil {
   fmt.Printf("消费者处理任务 %v 时发生 panic: %v\n", task, r)
  }
 }()

 if task.Handler != nil {
  err := task.Handler(task)
  if err != nil {
   fmt.Printf("消费者处理任务 %v 时发生错误: %v\n", task, err)
   return
  }

  fmt.Printf("消费者处理任务：%v 完成, 已消费：%d 个\n", task, pc.consumedCount.Load())
 }
}

// 启动一个异步生产者
func (pc *ProducerConsumer) StartProducer(numTask int) {
 go func() {
  // 每个生产者生产 numTask 个产品
  for n := range numTask {
   task := NewTask(n)
   pc.tasks <- task
   fmt.Printf("生产者生产任务：%v\n", task)
  }

  close(pc.tasks)
  fmt.Printf("生产者退出了\n")
 }()
}

// 等待消费者退出
func (pc *ProducerConsumer) Wait() {
 <-pc.consumerDone
 fmt.Printf("都退出了，消费总数：%d\n", pc.consumedCount.Load())
}

func main() {
 pc := NewProducerConsumer(10)
 pc.StartConsumer()
 pc.StartProducer(100)
 pc.Wait()
}
```

多个生产者和多个消费者，生产者异步生产完后就退出，主 Goroutine 等待所有生产者都退出后，关闭任务队列，这样消费者异步处理完所有任务后也退出，接着主 Goroutine 等待所有消费者退出，再退出。

```go

type ProducerConsumer struct {
 tasks       chan Task      // 生产者和消费者共享一个通道，不需要额外加锁
 wgProducers sync.WaitGroup // 生产者等待组，用于等待所有生产者退出
 wgConsumers sync.WaitGroup // 消费者等待组，用于等待所有消费者退出
}

func NewProducerConsumer(bufSize int) *ProducerConsumer {
 return &ProducerConsumer{
  tasks: make(chan Task, bufSize),
 }
}

// 启动 n 个异步消费者
func (pc *ProducerConsumer) StartConsumer(numConsumers int) {
 for i := range numConsumers {
  pc.wgConsumers.Add(1)
  go func(id int) {
   defer pc.wgConsumers.Done()

   for task := range pc.tasks {
    pc.handleTask(&task)
    fmt.Printf("消费者 %d 消费任务：%v\n", id, task)
   }

   fmt.Printf("消费者 %d 退出了\n", id)
  }(i)
 }
}

 defer func() {
  // 当 task 的 handler 中发生 panic 时，不至于让消费者协程退出
  if r := recover(); r != nil {
   fmt.Printf("消费者处理任务 %v 时发生 panic: %v\n", task, r)
  }
 }()

 if task.Handler != nil {
  err := task.Handler(task)
  if err != nil {
   fmt.Printf("消费者处理任务 %v 时发生错误: %v\n", task, err)
   return
  }

  fmt.Printf("消费者处理任务：%v 完成\n", task)
 }
}

// 启动 n 个异步生产者
func (pc *ProducerConsumer) StartProducer(numProducers, numTask int) {
 for i := range numProducers {
  pc.wgProducers.Add(1)

  go func(id int) {
   defer pc.wgProducers.Done()

   // 每个生产者生产 jobsPerProducer 个产品
   for n := range numTask {
    task := NewTask(n)
    pc.tasks <- task
    fmt.Printf("生产者 %d 生产任务：%v\n", id, task)
   }

   fmt.Printf("生产者 %d 退出了\n", id)
  }(i)
 }
}

// 等待生产者和消费者退出
func (pc *ProducerConsumer) Wait() {

 fmt.Printf("所有生产者都退出了\n")
 close(pc.tasks) // 关闭通道，通知所有消费者退出
 pc.wgConsumers.Wait()
 fmt.Printf("所有消费者都退出了\n")
 fmt.Printf("都退出了\n")
}

}
```

多个生产者和多个消费者，生产者和消费者都是**持续**运行（异步），生产者收到 quit 信号才退出（由 Goroutine 负责发出），主 Goroutine 等待所有生产者都退出后，关闭任务队列，这样消费者处理完所有任务也退出，接着主 Goroutine 等待所有消费者退出，再退出。

```go

func (t Task) String() string {

// 启动 n 个异步生产者

  pc.wgProducers.Add(1)

  go func(id int) {
   defer pc.wgProducers.Done()

   // 持续生产，直到收到 quit 信号
   for n := 0; ; n++ {
    task := Task{
     ID:      id*10000000 + n,
     Content: fmt.Sprintf("%d-%d", id, n),

    case <-pc.producterQuit:
     fmt.Printf("生产者 %d 收到 quit 信号，退出了\n", id)

     fmt.Printf("生产者 %d 生产任务：%v\n", id, task)
    }
   }
  }(i)
 }
}

 pc.wgProducers.Wait()
 fmt.Printf("所有生产者都退出了\n")
 close(pc.tasks) // 关闭通道，通知所有消费者退出
 pc.wgConsumers.Wait()
 fmt.Printf("所有消费者都退出了\n")
 fmt.Printf("都退出了\n")
}

}
```

### Worker Pool 模式

工作池模式：一组 worker 协程持续从一个任务队列中读取任务并处理（多个 worker 抢同一个任务队列），直到关闭任务通道才退出。

可以控制并发数量（工作池中的 worker 数量），复用协程，减少协程创建、销毁的开销。

```go
type Task struct {
 ID      int
 Content string
 Handler func(t *Task) error
}

func (t Task) String() string {
 return fmt.Sprintf("id:%d,content:%s", t.ID, t.Content)
}

func NewTask(id int) Task {
 return Task{
  ID:      id,
  Content: strconv.Itoa(id),
  Handler: func(t *Task) error {
   // time.Sleep(time.Duration(rand.IntN(1000)) * time.Millisecond) // 模拟业务逻辑

   if rand.IntN(2) == 0 {
    return fmt.Errorf("模拟失败")
   }

   if rand.IntN(2) == 0 {
    panic("模拟 panic")
   }

   return nil
  },
 }
}

// WorkerPool 工作池结构体
type WorkerPool struct {
 tasks chan Task      // 任务队列
 wg    sync.WaitGroup // 等待所有 worker 退出
}

// NewWorkerPool 创建工作池
func NewWorkerPool(numWorker int, bufSize int) *WorkerPool {
 pool := &WorkerPool{
  tasks: make(chan Task, bufSize),
 }

 // 启动固定数量 worker Goroutine
 for i := range numWorker {
  pool.wg.Go(func() { // 启动协程
   pool.worker(i)
  })
 }
 return pool
}

// worker 单个工作协程循环处理任务
func (p *WorkerPool) worker(workerID int) {
 for task := range p.tasks {
  p.handleTask(workerID, &task)
 }
 fmt.Printf("任务通道关闭，worker-%d，退出\n", workerID)
}

// handleTask 业务任务处理逻辑
func (p *WorkerPool) handleTask(workerID int, task *Task) {
 defer func() {
  // 当 task 的 handler 中发生 panic 时，不至于让消费者协程退出
  if r := recover(); r != nil {
   fmt.Printf("worker-%d 处理任务 %v 时发生 panic: %v\n", workerID, task, r)
  }
 }()

 fmt.Printf("worker-%d 开始处理任务: %v\n", workerID, task)

 if task.Handler != nil {
  err := task.Handler(task)
  if err != nil {
   fmt.Printf("worker-%d 处理任务 %v 失败: %v\n", workerID, task, err)
  } else {
   fmt.Printf("worker-%d 处理任务 %v 成功\n", workerID, task)
  }
 }

 fmt.Printf("worker-%d 完成任务处理：%v\n", workerID, task)
}

// Submit 提交任务到工作池（非阻塞，通道满会阻塞，超时便放弃）
func (p *WorkerPool) Submit(task *Task, timeout time.Duration) {
 select {
 case p.tasks <- *task:
  // 任务队列未满，直接提交
 case <-time.After(timeout):
  // 任务队列已满，超时未提交，放弃
  fmt.Printf("任务 %v 超时未提交，放弃\n", task)
 }
}

// Close 优雅关闭工作池
func (p *WorkerPool) Close() {
 close(p.tasks) // 关闭任务通道，worker 感知通道关闭自动退出
 p.wg.Wait()    // 等待所有 worker 退出
 fmt.Println("工作池已全部关闭")
}

func main() {
 // 创建池：5 个并发 worker，任务队列缓冲 10
 pool := NewWorkerPool(5, 10)

 // 提交 20 个任务
 for n := 1; n <= 20; n++ {
  task := NewTask(n)
  pool.Submit(&task, time.Duration(rand.IntN(1000))*time.Millisecond)
 }

 // 等待所有任务执行完成，关闭池
 pool.Close()
}
```

动态扩容、缩容：

- 原子计数当前运行 worker 数：sync/atomic 避免锁竞争
- 扩容：调用 Expand(n)，循环启动 n 个新 worker，不超过最大上限
- 缩容：调用 Shrink(n)，向 n 个 worker 发送退出信号，worker 处理完当前任务后退出。至少保留 1 个 worker
- worker 内部 select 同时监听任务通道、个人退出信号

自动扩缩容（无需手动调用 Expand/Shrink）：后台起一个监控协程，定时检测队列堆积长度，队列缓冲占用超过 80% -> 自动扩容；队列长时间为空，且当前 worker > 最小阈值 -> 自动缩容

### Pipeline 模式

多个协程通过通道连接起来，每个协程是一个处理阶段，上一阶段的输出作为下一阶段的输入，形成数据处理流水线，最后一个阶段的读操作会依次反向链式触发前面阶段的处理，形成 lazily 驱动的流水线。

最简单的 Pipeline 由一系列函数组成，每个函数接收一个只读通道，返回一个只读通道。

```go
// 第一阶段：生成数据
func generateNumbers(ctx context.Context, count int) <-chan int {
 out := make(chan int)
 go func() {
  defer close(out)
  for i := 1; i <= count; i++ {
   select {
   case out <- i:
   case <-ctx.Done(): // 监听取消事件
    return
   }
  }
 }()
 return out
}

// 第二阶段：平方计算
func squareNumbers(ctx context.Context, in <-chan int) <-chan int {
 out := make(chan int)
 go func() {
  defer close(out)
  for n := range in {
   select {
   case out <- n * n:
   case <-ctx.Done():
    return
   }
  }
 }()
 return out
}

// 第三阶段：过滤偶数
func filterEven(ctx context.Context, in <-chan int) <-chan int {
 out := make(chan int)
 go func() {
  defer close(out)
  for n := range in {
   if n%2 == 0 { // 只输出偶数结果
    select {
    case out <- n:
    case <-ctx.Done():
     return
    }
   }
  }
 }()
 return out
}

func main() {
 ctx, cancel := context.WithCancel(context.Background())
 defer cancel()

 // 链式调用：数据流从 generate -> square -> filter -> main
 for result := range filterEven(ctx, squareNumbers(ctx, generateNumbers(ctx, 10))) {
  fmt.Printf("%d\n", result)
 }
}
```

### Fan-out、Fan-in 模式

扇入是指将多个通道的数据合并到一个通道，相当于”Reduce“

扇出是指一个通道的数据被多个 worker 协程并发处理，相当于”Map“，每个 worker 有独立的输出通道。在生产环境中，通常采用分片（sharding）策略，将输入数据切分为 N 份，每个 worker 只认领其中一份，有两种方案：

1. 静态分片（Range Partitioning），适用场景：输入数据是已知的静态切片（如 []int），且每个任务的执行时间大致相同。实现方案：将切片按 worker 数量均匀切块（chunk）。
2. 动态负载均衡（Work-Stealing / 共享队列），适用场景：每个任务的处理耗时差异极大，静态分片会导致“木桶效应”——最慢的 worker 拖垮全流程。实现方案：引入一个共享的任务通道，让所有 worker 从这个通道里面抢任务，但输出结果依然通过 Fan-in 合并，这本质上是 Worker Pool + Pipeline 的混合体。

```go
// Worker Pool + Pipeline 的混合体，所有 worker 都从一个通道里面抢任务
type Task struct {
 ID      int
 Content string
 Handler func(t *Task) error
}

func (t Task) String() string {
 return fmt.Sprintf("id:%d,content:%s", t.ID, t.Content)
}

func NewTask(id int) Task {
 return Task{
  ID:      id,
  Content: strconv.Itoa(id),
  Handler: func(t *Task) error {
   // time.Sleep(time.Duration(rand.IntN(1000)) * time.Millisecond) // 模拟业务逻辑

   if rand.IntN(2) == 0 {
    return fmt.Errorf("模拟失败")
   }

   if rand.IntN(2) == 0 {
    panic("模拟 panic")
   }

   return nil
  },
 }
}

type Result struct {
 Content string
}

// 数据生产者：将切片转换为通道
func produceData(tasks []Task) <-chan Task {
 out := make(chan Task)
 go func() {
  defer close(out)
  for _, item := range tasks {
   out <- item
  }
 }()
 return out
}

// Fan-out：启动 N 个 worker 并发处理（工作池模式）
func FanOut(numWworker int, tasks []Task) []<-chan Result {
 // 创建任务通道，所有 worker 从这个通道抢任务
 input := produceData(tasks)

 // 每个 worker 拥有独立的输出通道
 outputs := make([]<-chan Result, numWworker)

 for i := range numWworker {
  out := make(chan Result)
  outputs[i] = out

  go func() {
   defer close(out)
   for task := range input {
    // 模拟处理时间
    time.Sleep(time.Millisecond * 100)
    processed := fmt.Sprintf("worker: %d，处理完成任务 %v", i, task)
    out <- Result{Content: processed}
   }
  }()
 }

 return outputs
}

// 合并多个 Channel（Fan-in）
func FanIn(Channels ...<-chan Result) <-chan Result {
 var wg sync.WaitGroup
 out := make(chan Result)

 // 从每个 Channel 读取数据
 output := func(c <-chan Result) {
  defer wg.Done()
  for data := range c {
   out <- data
  }
 }

 for _, c := range Channels {
  wg.Add(1)
  go output(c)
 }

 // 绝对不能直接在这里 close(out)，必须等待所有读取完成，由独立监控协程关闭
 // 也不能在这里直接 wg.Wait()，因为流水线是由最后一个阶段的读操作反向触发的，在这里阻塞主协程后，就没办法触发最后一个阶段的读操作了
 go func() {
  wg.Wait()
  close(out)
 }()

 return out
}

func main() {
 tasks := make([]Task, 0, 100)

 for i := range 100 {
  task := NewTask(i)
  tasks = append(tasks, task)
 }

 // Fan-out：启动多个 worker 处理数据，所有 worker 抢一个任务队列
 const numWorker = 10
 workersOutput := FanOut(numWorker, tasks)

 // Fan-in：合并所有 worker 的结果
 merged := FanIn(workersOutput...)

 // 消费合并后的结果
 for result := range merged {
  fmt.Println(result.Content)
 }

 fmt.Println("所有任务处理完成")
}
```

```go
// 在上例的基础上先做一次静态分片，让每个 worker 只处理一段数据
// 分片工具：将 tasks 切分为 numWorker 个块，每个 worker 处理其中的一个 chunk
func partition(tasks []Task, numWorker int) [][]Task {
 var chunks [][]Task
 chunkSize := (len(tasks) + numWorker - 1) / numWorker // 向上取整，避免遗漏
 for i := 0; i < len(tasks); i += chunkSize {
  end := min(i+chunkSize, len(tasks))
  chunks = append(chunks, tasks[i:end])
 }
 return chunks
}

// Fan-out（Partitioned）：每个 Worker 处理自己独有的分片
func FanOutPartitioned(numWworker int, tasks []Task) []<-chan Result {
 chunks := partition(tasks, numWworker)
 outputs := make([]<-chan Result, len(chunks)) // 实际启动的 Worker 数 = 分片数

 for i, chunk := range chunks {
  out := make(chan Result)
  outputs[i] = out

  go func() {
   defer close(out)
   for _, task := range chunk {
    // 模拟处理时间
    time.Sleep(time.Millisecond * 100)
    processed := fmt.Sprintf("worker: %d，处理完成任务 %v", i, task)
    out <- Result{Content: processed}
   }
  }()
 }
 return outputs
}

func main() {
 tasks := make([]Task, 0, 100)

 for i := range 100 {
  task := NewTask(i)
  tasks = append(tasks, task)
 }

 const numWorker = 10
 workersOutput := FanOutPartitioned(numWorker, tasks)

 // Fan-in：合并所有 worker 的结果
 merged := FanIn(workersOutput...)

 // 消费合并后的结果
 for result := range merged {
  fmt.Println(result.Content)
 }

 fmt.Println("所有任务处理完成")
}
```

### Pub-Sub 模式

Pub-Sub（发布 - 订阅）模式，本质上是在多个 订阅者（Subscriber）和 发布者（Publisher）之间通过一个消息代理（Broker）进行解耦。
  
```go
type Message struct {
 Topic   string
 Content any
}

type Broker struct {
 sync.RWMutex                                     // 保护下面的 map
 subscribers  map[string]map[*Subscriber]struct{} // map 的 key 为 topic，value 为订阅了这个 topic 的所有订阅者结构体 (其中包括用来接收发布消息的通道)
}

type Subscriber struct {
 C chan Message
}

func (s *Subscriber) OnMessage() {
 // 异步监听所订阅主题的消息，不影响主协程
 go func() {
  for msg := range s.C {
   fmt.Printf("topic: %s, msg: %v\n", msg.Topic, msg.Content)
  }
 }()
}

func NewBroker() *Broker {
 return &Broker{
  subscribers: make(map[string]map[*Subscriber]struct{}),
 }
}

// 订阅主题
func (b *Broker) Subscribe(topic string) *Subscriber {
 b.Lock()
 defer b.Unlock()

 subscriber := &Subscriber{
  C: make(chan Message, 10), // 每个订阅者都有一个独立通道，订阅者监听这个通道，就能收到发布的消息。注意：这里的实现是一个订阅者一个通道，发布一个消息后，所有的订阅者都能收到这个消息。如果所有的订阅者共用一个通道，那么发布一个消息后，就只有一个订阅者能收到这个消息。
 }

 if _, exists := b.subscribers[topic]; !exists {
  b.subscribers[topic] = make(map[*Subscriber]struct{})
 }
 b.subscribers[topic][subscriber] = struct{}{}

 return subscriber
}

// 发布消息
func (b *Broker) Publish(topic string, content any) {
 b.RLock()
 defer b.RUnlock()

 message := Message{Topic: topic, Content: content}
 for subscriber := range b.subscribers[topic] {
  // 异步发布消息，不影响主协程
  go func(s *Subscriber) {
   s.C <- message
  }(subscriber)
 }
}

// 取消订阅
func (b *Broker) Unsubscribe(topic string, subscriber *Subscriber) {
 b.Lock()
 defer b.Unlock()

 if _, exists := b.subscribers[topic]; exists {
  close(subscriber.C)
  delete(b.subscribers[topic], subscriber)
 }
}
func main() {
 broker := NewBroker()

 // 订阅者
 sub1 := broker.Subscribe("news")
 sub1.OnMessage()

 sub2 := broker.Subscribe("news")
 sub2.OnMessage()

 sub3 := broker.Subscribe("sports")
 sub3.OnMessage()

 // 随机发布消息
 topics := []string{"news", "sports"}
 for range 5 {
  broker.Publish(topics[rand.IntN(len(topics))], fmt.Sprintf("Message %d", rand.IntN(100)))
 }

 time.Sleep(time.Second) // 等待消息处理

 broker.Unsubscribe("news", sub1)
 broker.Unsubscribe("news", sub2)
 broker.Unsubscribe("sports", sub3)
}
```
