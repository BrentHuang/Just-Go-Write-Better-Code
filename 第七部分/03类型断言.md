# 类型断言（Type Assertion）

类型断言提供了访问接口变量持有的具体类型的具体值的途径。类型断言的操作对象是接口值，目的是检查接口持有的具体类型，并将其持有的具体值取出。

语句 `t := i.(T)` 断言接口变量 `i` 持有的具体类型为 `T`，并将 `i` 持有的 `T` 类型的值赋值给变量 `t`。如果 `i` 持有的具体类型不是 `T`，则表示断言失败，会引发 panic。

类型断言可以返回两个值：接口持有的具体值，以及一个用于判断断言是否成功的 `bool` 值 `ok`。`ok` 语法不会引发 panic，实际项目中应始终使用这种语法。

```go
t, ok := i.(T) // 如果 i 持有的具体类型不是 T，断言失败，ok 为 false，t 为类型 T 的零值；如果 i 持有的具体类型是 T，断言成功，ok 为 true，t 为 i 持有的具体类型 T 的具体值
```

```go
func main() {
 var i interface{} = "hello"

 s := i.(string)
 fmt.Println(s) // hello

 s, ok := i.(string)
 fmt.Printf("%s, %t\n", s, ok) // hello, true

 // f := i.(float64) // panic -- 断言失败
 // fmt.Println(f)

 f, ok := i.(float64)
 fmt.Printf("%f, %t\n", f, ok) // 0.000000, false
}
```

注意：

- 类型断言的操作对象 `i` 必须是一个接口值
- 直接对 `nil` 接口断言会引发 panic，`ok` 语法对 `nil` 接口断言不会引发 panic，但 `ok` 为 false

```go
func main() {
 var i any
 // a := i.(int) // panic

 a, ok := i.(int)
 fmt.Printf("%v, %t\n", a, ok) // 0, false
}
```

---

类型断言要求具体类型完全匹配，`T` 和 `*T` 是两个不同的类型。如果目标类型根本没实现接口，编译器直接报错（编译期检查）；如果目标类型实现了接口但具体类型不匹配，则运行时失败（`ok` 为 false 或 panic）。

```go
type I interface {
 M()
}

type MyStruct1 struct {
}

type MyStruct2 struct {
}

// 方法接收者参数为值类型，MyStruct1 和 *MyStruct1 都实现了接口，MyStruct1 类型的值变量或 *MyStruct1 类型的指针变量均可赋值给接口 I
func (m MyStruct1) M() {
 println("MyStruct1.M")
}

// 方法接收者参数为指针类型，仅 *MyStruct2 实现了接口，MyStruct2 没有实现接口，仅 *MyStruct2 类型的指针变量可赋值给接口 I
func (m *MyStruct2) M() {
 println("MyStruct2.M")
}

func main() {
 var i I

 {
  // m := i.(MyStruct1) // 直接对 nil 接口断言会引发 panic（非 ok 语法）
  // fmt.Println(m)

  m1, ok := i.(MyStruct1) // 对 nil 接口断言不会 panic，但 ok 必为 false
  fmt.Printf("%v, %t\n", m1, ok)     // {}, false

  m2, ok := i.(*MyStruct1) // 对 nil 接口断言不会 panic，但 ok 必为 false
  fmt.Printf("%v, %t\n", m2, ok)      // <nil>, false

  // m3, ok := i.(MyStruct2) // MyStruct2 没有实现接口，不能断言 MyStruct2，编译报错：MyStruct2 does not implement I (method M has pointer receiver)
  // fmt.Printf("%v, %t\n", m3, ok)

  m4, ok := i.(*MyStruct2) // 对 nil 接口断言不会 panic，但 ok 必为 false
  fmt.Printf("%v, %t\n", m4, ok)      // <nil>, false
 }

 {
  s1 := MyStruct1{}

  i = s1 // 将 MyStruct1 的值变量赋值给接口，断言 MyStruct1 类型成功，断言 *MyStruct1 类型失败
  m1, ok := i.(MyStruct1)
  fmt.Printf("%v, %t\n", m1, ok) // {}, true

  m2, ok := i.(*MyStruct1)
  fmt.Printf("%v, %t\n", m2, ok) // <nil>, false

  i = &s1 // 将 *MyStruct1 的指针变量赋值给接口，断言 MyStruct1 类型失败，断言 *MyStruct1 类型成功
  m3, ok := i.(MyStruct1)
  fmt.Printf("%v, %t\n", m3, ok) // {}, false

  m4, ok := i.(*MyStruct1)
  fmt.Printf("%v, %t\n", m4, ok) // &{}, true
 }

 {
  s2 := MyStruct2{}

  // i = s2 // MyStruct2 没有实现接口，不能将 MyStruct2 类型的变量赋值给接口 I，编译报错
  // m1, ok := i.(MyStruct2) // MyStruct2 没有实现接口，不能断言 MyStruct2，编译报错
  // fmt.Printf("%v, %t\n", m1, ok)

  m2, ok := i.(*MyStruct2) // 此时 i 为 nil
  fmt.Printf("%v, %t\n", m2, ok)      // <nil>, false

  i = &s2
  // m3, ok := i.(MyStruct2) // MyStruct2 没有实现接口，不能断言 MyStruct2，编译报错
  // fmt.Printf("%v, %t\n", m3, ok)

  m4, ok := i.(*MyStruct2)
  fmt.Printf("%v, %t\n", m4, ok) // &{}, true
 }
}
```

错误处理中，`errors.As` 本质上是类型断言的泛型封装。

## 断言接口类型

类型断言的 `T` 不仅可以是具体类型，也可以是接口类型。例如 `i.(io.Reader)` 断言 `i` 持有的具体类型是否实现了 `io.Reader` 接口，断言成功的结果类型是 `io.Reader` 接口类型。

```go
type MyReader struct{}

func (r *MyReader) Read(p []byte) (n int, err error) {
 return 0, nil
}

func main() {
 var i any
 i = &MyReader{}

 // 断言为具体类型 *MyReader
 r, ok := i.(*MyReader)
 fmt.Printf("*MyReader: %v, ok=%t\n", r, ok) // *MyReader: &{}, ok=true

 // 断言为接口类型 io.Reader —— i 持有的 *MyReader 实现了 io.Reader，断言成功
 // 注意：r2 的类型是 io.Reader（接口类型），不是 *MyReader
 r2, ok := i.(io.Reader)
 fmt.Printf("io.Reader: %v, ok=%t\n", r2, ok) // io.Reader: &{}, ok=true
 // 通过 r2 只能调用 io.Reader 的方法（Read），无法调用 Close

 // 断言为没有实现的接口，ok 为 false
 w, ok := i.(io.Writer)
 fmt.Printf("io.Writer: %v, ok=%t\n", w, ok) // io.Writer: <nil>, ok=false
}
```
