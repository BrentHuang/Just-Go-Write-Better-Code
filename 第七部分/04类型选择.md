# 类型选择（Type Switch）

类型选择与一般的 switch-case 语句相似，但 case 为指定类型（而非值/表达式），操作对象是接口值，目的是检查接口持有的具体类型并执行不同的操作。

类型选择与类型断言 `i.(T)` 语法相同，只是将具体类型 `T` 替换为关键字 `type`。

```go
switch v := i.(type) { // 不会 panic
case T:
 // 如果接口 i 持有的具体类型是 T，则走到这个分支。v 为 T 类型的具体值
case S:
 // 如果接口 i 持有的具体类型是 S，则走到这个分支。v 为 S 类型的具体值
case nil:
 // 如果接口 i 为 nil，则走到这个分支
  fmt.Printf("(%[1]T, %[1]v)\n", i) // (<nil>, <nil>)
default:
 // 未匹配到 case 分支中的任何类型，则走到 default 分支。在 default 分支中，v 为接口类型
 fmt.Printf("%v\n", v)
}
```

完整示例：

```go
func do(i interface{}) {
 fmt.Printf("(%[1]T, %[1]v)\n", i)

 switch v := i.(type) {
 case []int: // 处理 int 切片
  fmt.Printf("i 持有的具体类型是 []int，长度：%d\n", len(v))
 case map[string]int: // 处理 map
  fmt.Printf("i 持有的具体类型是 map[string]int，键值对数量：%d\n", len(v))
 case int, float64: // 合并判断数字类型
  fmt.Printf("i 持有的具体类型是 int 或 float64\n") // 在多类型 case 分支中，v 为接口类型
  if a, ok := v.(int); ok {
   fmt.Printf("i 持有的具体类型是 int：%v\n", a)
  } else {
   a := v.(float64)
   fmt.Printf("i 持有的具体类型是 float64：%v\n", a)
  }
  // fallthrough 编译报错
 case string: // 处理字符串
  fmt.Printf("i 持有的具体类型是 string：%v\n", v)
 case nil: // 接口 i 为 nil
  fmt.Println("接口 i 为 nil")
 default:
  fmt.Printf("未知类型\n") // 在 default 分支中，v 为接口类型
  if a, ok := v.(bool); ok {
   fmt.Printf("i 持有的具体类型是 bool：%v\n", a)
  }
 }
}

func main() {
 var i interface{}
 do(i) // 匹配 case nil 分支

 var p *int
 i = p
 do(i) // 走到 default 分支 (*int, <nil>)

 do([]int{1, 2, 3})                 // 匹配 case []int 分支
 do(map[string]int{"a": 1, "b": 2}) // 匹配 case map[string]int 分支
 do(42)                             // 匹配 case int 分支
 do(3.14)                           // 匹配 case float64 分支
 do("hello")                        // 匹配 case string 分支
 do(true)                           // 走到 default 分支
}
```

注意：

- 类型选择的操作对象 `i` 必须是一个接口值
- 类型选择本身包含安全检查，不会引发 panic，没有像类型断言那样的 ok 语法
- case 后是类型而非值/表达式，一个 case 可以匹配多个类型（它们的处理逻辑一样）
- 单类型 case 分支中 `v` 已是该具体类型的具体值，可直接使用；多类型 case 与 default 分支中 `v` 仍为接口类型，如需使用具体类型的具体值还需要再次断言
- 不能使用 fallthrough，编译报错

case 后也可以是接口类型，如果接口持有的具体类型实现了该接口，就会走到这个分支。

```go
// 对比具体类型和接口类型两种 case
type MyReader struct{}

func (MyReader) Read(p []byte) (int, error) { return 0, nil }

func main() {
 var i any = MyReader{}

 switch v := i.(type) {
 case io.Reader: // 如果接口 i 持有的具体类型（MyReader）实现了 io.Reader，则走到这个分支
  fmt.Printf("(%[1]T, %[1]v)\n", v) // v 的类型是 io.Reader（接口类型）
 case MyReader: // 如果接口 i 持有的具体类型是 MyReader，则走到这个分支
  fmt.Printf("%v\n", v) // v 的类型是 MyReader（具体类型）
 }
}
```

---

若类型选择的操作对象 `i` 是特定接口类型而非 `interface{}`/`any`，则 case 后的类型必须实现了该特定接口，否则编译报错。

```go
type I interface {
 M()
}

type T1 struct {
}

type T2 struct {
}

// 方法接收者参数类型为 *T1，只有 *T1 实现了接口，T1 没有实现接口
func (t *T1) M() {
 println("T1.M")
}

// 方法接收者参数类型为 *T2，只有 *T2 实现了接口，T2 没有实现接口
func (t *T2) M() {
 println("T2.M")
}

func checkType(i I) {
 switch v := i.(type) {
 case *T1:
  v.M()
 case *T2:
  v.M()
 // case T1: // 编译报错：不可能的分支：i 持有的具体类型不可能是 T1（因为 T1 没有实现 I）
 //  println("T1")
 // case T2: // 编译报错：不可能的分支：i 持有的具体类型不可能是 T2（因为 T2 没有实现 I）
 //  println("T2")
 // case int: // 编译报错：不可能的分支：i 持有的具体类型不可能是 int（因为 int 没有实现 I）
 //  println("是 int 值")
 case nil:
  println("接口变量为 nil")
 default:
  println("未知类型")
 }
}

func main() {
 var i I
 checkType(i)
 checkType(&T1{})
 checkType(&T2{})

 // checkType(T1{}) // T1 没有实现接口 I，不能将 T1 类型的变量赋值给 i。编译报错
 // checkType(T2{}) // T2 没有实现接口 I，不能将 T2 类型的变量赋值给 i。编译报错
 // checkType(1)    // int 类型没有实现接口 I，不能将 int 类型的变量赋值给 i。编译报错
}
```

---

类型选择与类型断言可以互相转换，以下两个代码段等价：

```go
// 类型选择
func f(x interface{}) {
 switch x.(type) {
 case int, uint: // ...
 case bool: // ...
 case string: // ...
 case nil: // ...
 default: // ...
 }
}

// 类型断言
func g(x interface{}) {
 if x == nil {
  // ...
 } else if _, ok := x.(int); ok {
  // ...
 } else if _, ok := x.(uint); ok {
  // ...
 } else if _, ok := x.(bool); ok {
  // ...
 } else if _, ok := x.(string); ok {
  // ...
 } else {
  // ...
 }
}
```
